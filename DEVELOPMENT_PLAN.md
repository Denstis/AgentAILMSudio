## 🚀 Быстрый старт

```python
# Использование LM Studio API
from lm_agent.api.lmstudio import LMStudioClient

client = LMStudioClient(base_url="http://localhost:1234/v1")

# Получить список моделей
models = client.list_models()
for model in models:
    print(f"{model.name}: контекст={model.context_window}")

# Использовать систему ролей
from lm_agent.core.roles import RoleEngine

engine = RoleEngine()
engine.set_active_role('code_generator')

# Построить сообщения с учётом роли
messages = engine.build_messages('Напиши функцию сортировки')

# Отправить запрос
response = client.chat_completion(
    model=models[0].id,
    messages=messages,
    temperature=0.7
)
print(response['choices'][0]['message']['content'])
```
